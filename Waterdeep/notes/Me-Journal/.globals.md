```ui
// Main stats
character_name("Magister Septimus el Blau")
background("Charlatan")
class_("Sorcerer")
species("Gnome")
subclass("Wild Magic")
level(3)

// Ability scores
strength(8)
dexterity(16)
constitution(12)
intelligence(16)
wisdom(10)
charisma(20)

// Ability modifiers
// Ability modifiersf
strength_mod(((strength - 10) - ((strength - 10) % 2)) / 2)
dexterity_mod(((dexterity - 10) - ((dexterity - 10) % 2)) / 2)
constitution_mod(((constitution - 10) - ((constitution - 10) % 2)) / 2)
intelligence_mod(((intelligence - 10) - ((intelligence - 10) % 2)) / 2)
wisdom_mod(((wisdom - 10) - ((wisdom - 10) % 2)) / 2)
charisma_mod(((charisma - 10) - ((charisma - 10) % 2)) / 2)

// Proficiency
proficiency_bonus(
  level >= 17 ? 6 :
  level >= 13 ? 5 :
  level >= 9  ? 4 :
  level >= 5  ? 3 :
                2
)

// Combat
base_armor_class(10)
shield_equipped(false)
shield_bonus(0)
armor_class(base_armor_class + dexterity_mod + shield_bonus)

hp_current(17)
hp_temp(0)
hp_max(
  (6 + constitution_mod) +
  ((level - 1) * (4 + constitution_mod))
)

hit_dice_spent(0)
hit_dice_max(level)

initiative(dexterity_mod)
speed(30)
size("Small")
passive_perception(10 + wisdom_mod)

// Saving throw proficiencies
save_prof_strength(false)
save_prof_dexterity(false)
save_prof_constitution(true)
save_prof_intelligence(false)
save_prof_wisdom(false)
save_prof_charisma(true)

save_strength(strength_mod + (save_prof_strength ? proficiency_bonus : 0))
save_dexterity(dexterity_mod + (save_prof_dexterity ? proficiency_bonus : 0))
save_constitution(constitution_mod + (save_prof_constitution ? proficiency_bonus : 0))
save_intelligence(intelligence_mod + (save_prof_intelligence ? proficiency_bonus : 0))
save_wisdom(wisdom_mod + (save_prof_wisdom ? proficiency_bonus : 0))
save_charisma(charisma_mod + (save_prof_charisma ? proficiency_bonus : 0))

// Skill proficiencies
acrobatics_prof(false)
animal_handling_prof(false)
arcana_prof(true)
athletics_prof(false)
deception_prof(true)
history_prof(false)
insight_prof(true)
intimidation_prof(false)
investigation_prof(true)
medicine_prof(false)
nature_prof(false)
perception_prof(true)
performance_prof(true)
persuasion_prof(false)
religion_prof(false)
sleight_of_hand_prof(true)
stealth_prof(false)
survival_prof(false)

// Skills
acrobatics(dexterity_mod + (acrobatics_prof ? proficiency_bonus : 0))
animal_handling(wisdom_mod + (animal_handling_prof ? proficiency_bonus : 0))
arcana(intelligence_mod + (arcana_prof ? proficiency_bonus : 0))
athletics(strength_mod + (athletics_prof ? proficiency_bonus : 0))
deception(charisma_mod + (deception_prof ? proficiency_bonus : 0))
history(intelligence_mod + (history_prof ? proficiency_bonus : 0))
insight(wisdom_mod + (insight_prof ? proficiency_bonus : 0))
intimidation(charisma_mod + (intimidation_prof ? proficiency_bonus : 0))
investigation(intelligence_mod + (investigation_prof ? proficiency_bonus : 0))
medicine(wisdom_mod + (medicine_prof ? proficiency_bonus : 0))
nature(intelligence_mod + (nature_prof ? proficiency_bonus : 0))
perception(wisdom_mod + (perception_prof ? proficiency_bonus : 0))
performance(charisma_mod + (performance_prof ? proficiency_bonus : 0))
persuasion(charisma_mod + (persuasion_prof ? proficiency_bonus : 0))
religion(intelligence_mod + (religion_prof ? proficiency_bonus : 0))
sleight_of_hand(dexterity_mod + (sleight_of_hand_prof ? proficiency_bonus : 0))
stealth(dexterity_mod + (stealth_prof ? proficiency_bonus : 0))
survival(wisdom_mod + (survival_prof ? proficiency_bonus : 0))

// Species rules
if(shield_equipped) {
  shield_bonus = 2
}
if(!shield_equipped) {
  shield_bonus = 0
}

if(species == "Human") {
  speed = 30
  size = "Medium"
}
if(species == "Elf") {
  speed = 30
  size = "Medium"
}
if(species == "Dwarf") {
  speed = 25
  size = "Medium"
}
if(species == "Halfling") {
  speed = 25
  size = "Small"
}
if(species == "Gnome") {
  speed = 30
  size = "Small"
}

// Spellcasting
spellcasting_ability("Charisma")
spell_attack_bonus(charisma_mod + proficiency_bonus)
spell_save_dc(8 + charisma_mod + proficiency_bonus)

// Spell slots
spell_slot_total([4, 2, 0, 0, 0])
spell_slot_expended([1, 0, 0, 0, 0])

spell_slot_remaining([
  spell_slot_total[0] - spell_slot_expended[0],
  spell_slot_total[1] - spell_slot_expended[1],
  spell_slot_total[2] - spell_slot_expended[2],
  spell_slot_total[3] - spell_slot_expended[3],
  spell_slot_total[4] - spell_slot_expended[4]
])

sorcery_points(2)
sorcery_points_max(2)
```
